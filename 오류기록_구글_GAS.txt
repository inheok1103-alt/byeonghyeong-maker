// === AI 제작기 '공유' 오류·회의 DB (Google Apps Script) ===
// 모든 사용자의 오류·회의가 하나의 구글시트에 누적되고, 누적 개선지침을 다시 내려줍니다.
// 1) 구글 시트 새로 만들기 → [확장] → [Apps Script] → 아래 코드 붙여넣기
// 2) [배포] → [새 배포] → 웹 앱 → 실행: 나 / 액세스: "모든 사용자" → URL 복사
// 3) 레포의 config.json 의 "logUrl" 에 그 URL 을 넣고 커밋 → 모두에게 공유됨
function doPost(e){
  try{
    var ss=SpreadsheetApp.getActiveSpreadsheet();
    var sh=ss.getSheetByName('log')||ss.insertSheet('log');
    if(sh.getLastRow()===0) sh.appendRow(['시각','종류','유형','단계','상세','개선지침','회의']);
    var d=JSON.parse(e.postData.contents);
    sh.appendRow([new Date(), d.kind||'', d.type||'', d.t||'', d.detail||'', d.hint||'', JSON.stringify(d.discussion||'')]);
    return ContentService.createTextOutput('ok');
  }catch(err){ return ContentService.createTextOutput('err: '+err); }
}
// hints=1 로 GET 하면 최근 회의 개선지침을 JSON으로 반환 → 모두의 출제에 반영
function doGet(e){
  try{
    var sh=SpreadsheetApp.getActiveSpreadsheet().getSheetByName('log');
    var hints=[];
    if(sh){ var v=sh.getDataRange().getValues();
      for(var i=v.length-1;i>0 && hints.length<10;i--){ if(v[i][1]==='meeting' && v[i][5]) hints.push(String(v[i][5])); } }
    return ContentService.createTextOutput(JSON.stringify({hints:hints})).setMimeType(ContentService.MimeType.JSON);
  }catch(err){ return ContentService.createTextOutput(JSON.stringify({hints:[],error:String(err)})).setMimeType(ContentService.MimeType.JSON); }
}
